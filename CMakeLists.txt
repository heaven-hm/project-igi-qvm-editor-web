cmake_minimum_required(VERSION 3.20)
project(igi_qvm_online VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(igi_qvm_core
    core/qvm_parser.cpp core/qvm_decompiler.cpp core/qvm_compiler.cpp
    core/qsc_lexer.cpp core/qsc_parser.cpp core/api.cpp)
target_include_directories(igi_qvm_core PUBLIC core)
if(EMSCRIPTEN)
    target_compile_options(igi_qvm_core PRIVATE -fexceptions)
    add_subdirectory(wasm)
else()
    add_executable(qvm_native tools/native.cpp)
    target_link_libraries(qvm_native PRIVATE igi_qvm_core)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(qvm_core_tests tests/core_tests.cpp)
        target_link_libraries(qvm_core_tests PRIVATE igi_qvm_core)
        target_compile_definitions(qvm_core_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        add_test(NAME qvm_core COMMAND qvm_core_tests)
    endif()
endif()
